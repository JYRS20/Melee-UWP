cmake_minimum_required(VERSION 3.25)
set(CMAKE_SYSTEM_NAME WindowsStore)
set(CMAKE_SYSTEM_VERSION 10.0)
project(melee_uwp_abi_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(MELEE_GCC_OBJECT_DIR "" CACHE PATH "MinGW GCC probe object directory")
add_library(melee-uwp-runtime STATIC ../uwp/mingw_runtime.cpp)
target_compile_options(melee-uwp-runtime PRIVATE /EHsc /W4)
target_link_options(melee-uwp-runtime INTERFACE /alternatename:___chkstk_ms=__chkstk)
foreach(unit endian_probe random)
    if(NOT EXISTS "${MELEE_GCC_OBJECT_DIR}/${unit}.obj")
        message(FATAL_ERROR "Compile the GCC probe objects first: ${unit}.obj")
    endif()
endforeach()
set(probe_objects "${MELEE_GCC_OBJECT_DIR}/endian_probe.obj" "${MELEE_GCC_OBJECT_DIR}/random.obj")
set_source_files_properties(${probe_objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_executable(melee-uwp-abi-test abi_main.cpp ${probe_objects})
target_compile_definitions(melee-uwp-abi-test PRIVATE WINAPI_FAMILY=WINAPI_FAMILY_APP _CRT_APP)
target_compile_options(melee-uwp-abi-test PRIVATE /EHsc /W4 /fp:strict)
target_link_options(melee-uwp-abi-test PRIVATE /APPCONTAINER)
target_link_libraries(melee-uwp-abi-test PRIVATE melee-uwp-runtime WindowsApp)
enable_testing()
add_test(NAME melee_uwp_abi COMMAND melee-uwp-abi-test)

add_executable(melee-uwp-foundation WIN32 abi_app.cpp abi_main.cpp ../uwp/storage.cpp ${probe_objects})
target_compile_definitions(melee-uwp-foundation PRIVATE MELEE_UWP_ABI_APP WINAPI_FAMILY=WINAPI_FAMILY_APP _CRT_APP NOMINMAX UNICODE _UNICODE)
target_compile_options(melee-uwp-foundation PRIVATE /EHsc /W4 /fp:strict)
target_link_options(melee-uwp-foundation PRIVATE /APPCONTAINER)
target_link_libraries(melee-uwp-foundation PRIVATE melee-uwp-runtime WindowsApp)
set(MELEE_UWP_PUBLISHER "CN=WWHDRecompDevelopment" CACHE STRING "Local test signing publisher")
configure_file(Package.appxmanifest.in AppxManifest.xml @ONLY)
