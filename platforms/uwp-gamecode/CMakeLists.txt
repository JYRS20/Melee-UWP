cmake_minimum_required(VERSION 3.25)
set(CMAKE_SYSTEM_NAME WindowsStore)
set(CMAKE_SYSTEM_VERSION 10.0)
project(melee_uwp_gamecode LANGUAGES CXX)
get_filename_component(MELEE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(MELEE_GCC_BUILD "" CACHE PATH "Directory containing verified GCC COFF objects and result.json")
if(NOT EXISTS "${MELEE_GCC_BUILD}/result.json")
    message(FATAL_ERROR "Compile and verify the original GCC objects before archiving")
endif()
file(READ "${MELEE_GCC_BUILD}/result.json" report)
string(JSON unit_count GET "${report}" units)
string(JSON passed GET "${report}" passed)
if(NOT unit_count EQUAL passed)
    message(FATAL_ERROR "GCC compilation did not pass for every translation unit")
endif()
file(STRINGS "${MELEE_ROOT}/cmake/game_sources.manifest" expected)
list(FILTER expected EXCLUDE REGEX "baselib/debugconsole_main\\.c$")
list(APPEND expected src/pc/vtxarray.c src/pc/widescreen.c)
list(SORT expected)
string(JSON object_count LENGTH "${report}" objects)
if(NOT object_count EQUAL unit_count)
    message(FATAL_ERROR "Invalid GCC object report")
endif()
math(EXPR last "${object_count}-1")
set(actual)
set(objects)
foreach(index RANGE 0 ${last})
    string(JSON source GET "${report}" objects ${index} source)
    string(JSON status GET "${report}" objects ${index} exit_code)
    if(NOT status EQUAL 0)
        message(FATAL_ERROR "Failed GCC object: ${source}")
    endif()
    string(REPLACE "/" "_" object_name "${source}")
    set(object "${MELEE_GCC_BUILD}/objects/${object_name}.obj")
    if(NOT EXISTS "${object}")
        message(FATAL_ERROR "Missing GCC object: ${source}")
    endif()
    list(APPEND actual "${source}")
    list(APPEND objects "${object}")
endforeach()
list(SORT actual)
if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "GCC object sources differ from the upstream manifest")
endif()
set_source_files_properties(${objects} PROPERTIES EXTERNAL_OBJECT TRUE)
add_library(melee-uwp-gamecode STATIC ${objects} ../uwp/mingw_runtime.cpp)
set_target_properties(melee-uwp-gamecode PROPERTIES LINKER_LANGUAGE CXX)

target_compile_options(melee-uwp-gamecode PRIVATE /EHsc /W4)
target_link_options(melee-uwp-gamecode INTERFACE /alternatename:___chkstk_ms=__chkstk)
